# Task persistence: SQLite via SQLAlchemy. Single-file DB, no server, no
# extra process. Wired to a single background worker thread (see backend/

sqlalchemy>=2.0

# Optional 4-bit quantization (enabled via FLUX_QUANT=4bit env var).
# Drops Flux Kontext/Fill transformer from ~12 GB to ~3.5 GB and T5-XXL
# from ~9 GB to ~3 GB so the whole pipeline fits in 16 GB VRAM without
# cpu_offload — eliminates PCIe thrashing on RTX 5080/16 GB.
bitsandbytes>=0.49.0

# Style LoRA loading via diffusers' multi-adapter API (see backend/loras.py).
peft>=0.13.0

# Florence-2 vision backend (backend/vision.py) for auto-masking, captions,
# and object detection. einops + timm are runtime deps of the custom
# modeling code that ships in the Florence-2 HF repo (loaded via
# trust_remote_code=True).
einops>=0.8.0
timm>=1.0.0

# Real-ESRGAN learned upscaler (backend/upscale.py). spandrel parses the
# .pth weights and instantiates the RRDBNet architecture without needing
# basicsr — much lighter dep tree than xinntao's original realesrgan
# package. Same loader ComfyUI uses for arbitrary super-res models.
spandrel>=0.4.0

# Background removal (backend/bg_remove.py). rembg bundles BiRefNet /
# u2net / isnet weights as ONNX files and runs them via onnxruntime —
# avoids the transformers ``trust_remote_code`` path that the upstream
# RMBG-2.0 / BiRefNet repos require. Default model is "birefnet-general"
# for high-quality alpha edges; override via AIPIC_BG_REMOVE_MODEL.
rembg>=2.0.50

# PuLID face-identity preservation (backend/pulid.py). insightface
# provides RetinaFace (detection) + ArcFace (512-dim embedding) for
# extracting the user's face vector. Runs on ONNX runtime (already
# pulled in by rembg). Only required when mode="pulid" is used.
insightface>=0.7.3

# Video generation (backend/video.py). LTX-Video + Wan 2.2 are both
# wired through diffusers; MP4 encoding goes through
# ``diffusers.utils.export_to_video`` which fans out to imageio[ffmpeg].
# The optional [ffmpeg] extra ships a bundled ffmpeg binary so users
# don't need a system install. Only required when mode="video" is used.
imageio[ffmpeg]>=2.34.0
